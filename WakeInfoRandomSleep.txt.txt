#include <Wire.h>

#define ATECC_ADDR        0x60
#define ATECC_WORD_CMD    0x03
#define ATECC_WORD_SLEEP  0x01

#define ATECC_MAX_RESPONSE  96
#define WIRE_BUFFER_SIZE    32

//-----------------------------------------------------
// CryptoAuth CRC-16
// Polynomial 0x8005
//-----------------------------------------------------
void atcaCrc(const uint8_t *data, size_t length, uint8_t crc[2])
{
    uint16_t crcRegister = 0;
    const uint16_t polynomial = 0x8005;

    for (size_t counter = 0; counter < length; counter++)
    {
        for (uint8_t mask = 0x01; mask != 0; mask <<= 1)
        {
            const uint8_t dataBit =
                (data[counter] & mask) ? 1 : 0;

            const uint8_t crcBit =
                (uint8_t)((crcRegister >> 15) & 0x01);

            crcRegister <<= 1;

            if (dataBit != crcBit)
            {
                crcRegister ^= polynomial;
            }
        }
    }

    crc[0] = (uint8_t)(crcRegister & 0xFF);
    crc[1] = (uint8_t)((crcRegister >> 8) & 0xFF);
}

//-----------------------------------------------------
// Generic command with no data field
//-----------------------------------------------------
void buildCommand(uint8_t opcode,
                  uint8_t param1,
                  uint16_t param2,
                  uint8_t packet[8])
{
    packet[0] = ATECC_WORD_CMD;
    packet[1] = 0x07;                 // Count
    packet[2] = opcode;
    packet[3] = param1;
    packet[4] = (uint8_t)(param2);
    packet[5] = (uint8_t)(param2 >> 8);

    atcaCrc(&packet[1], 5, &packet[6]);
}

//-----------------------------------------------------

void printHex(uint8_t value)
{
    if (value < 0x10)
    {
        Serial.print('0');
    }

    Serial.print(value, HEX);
}

void printBuffer(const uint8_t *buffer, size_t length)
{
    for (size_t i = 0; i < length; i++)
    {
        printHex(buffer[i]);
        Serial.print(' ');
    }

    Serial.println();
}

//-----------------------------------------------------
// Generate wake token
//-----------------------------------------------------
bool wakeChip()
{
    /*
       A CryptoAuth wake token is a sufficiently long low pulse on SDA.

       Temporarily stop Wire before directly controlling SDA.
    */

    Wire.end();

    pinMode(SDA, OUTPUT);
    digitalWrite(SDA, LOW);
    delayMicroseconds(80);

    pinMode(SDA, INPUT_PULLUP);

    // ATECC608B wake-high delay is at least 1.5 ms
    delayMicroseconds(2500);

    Wire.begin();
    Wire.setClock(100000);

    /*
       The device now has a 4-byte wake response waiting.
    */

    uint8_t wakeResponse[4];
    uint8_t received = Wire.requestFrom(
        (uint8_t)ATECC_ADDR,
        (uint8_t)sizeof(wakeResponse)
    );

    if (received != sizeof(wakeResponse))
    {
        while (Wire.available())
        {
            Wire.read();
        }

        Serial.println(F("Wake response not received."));
        return false;
    }

    for (uint8_t i = 0; i < sizeof(wakeResponse); i++)
    {
        wakeResponse[i] = Wire.read();
    }

    Serial.print(F("Wake response: "));
    printBuffer(wakeResponse, sizeof(wakeResponse));

    // Normal wake response: 04 11 33 43
    if (wakeResponse[0] != 0x04)
    {
        Serial.println(F("Invalid wake response length."));
        return false;
    }

    return true;
}

//-----------------------------------------------------

void sleepChip()
{
    Wire.beginTransmission(ATECC_ADDR);
    Wire.write(ATECC_WORD_SLEEP);

    uint8_t result = Wire.endTransmission();

    if (result != 0)
    {
        Serial.print(F("Sleep transmission failed: "));
        Serial.println(result);
    }
}

//-----------------------------------------------------
// Decode four-byte device status packets
//-----------------------------------------------------
void printStatusCode(uint8_t status)
{
    Serial.print(F("Device status: 0x"));
    printHex(status);
    Serial.print(F(" - "));

    switch (status)
    {
        case 0x00:
            Serial.println(F("Success"));
            break;

        case 0x01:
            Serial.println(F("CheckMac or Verify miscompare"));
            break;

        case 0x03:
            Serial.println(F("Parse error"));
            break;

        case 0x05:
            Serial.println(F("ECC fault"));
            break;

        case 0x07:
            Serial.println(F("Self-test error"));
            break;

        case 0x08:
            Serial.println(F("Health-test error"));
            break;

        case 0x0F:
            Serial.println(F("Execution error"));
            break;

        case 0x11:
            Serial.println(F("Wake response"));
            break;

        case 0xEE:
            Serial.println(F("Watchdog warning"));
            break;

        case 0xFF:
            Serial.println(F("CRC or communication error"));
            break;

        default:
            Serial.println(F("Unknown status"));
            break;
    }
}

//-----------------------------------------------------
// Validate response CRC
//-----------------------------------------------------
bool validateResponseCrc(const uint8_t *response, size_t length)
{
    if (length < 4)
    {
        return false;
    }

    uint8_t calculatedCrc[2];

    /*
       CRC includes count and response data/status,
       but excludes the final two CRC bytes.
    */
    atcaCrc(response, length - 2, calculatedCrc);

    const uint8_t receivedCrcLow  = response[length - 2];
    const uint8_t receivedCrcHigh = response[length - 1];

    Serial.print(F("Calculated CRC: "));
    printHex(calculatedCrc[0]);
    Serial.print(' ');
    printHex(calculatedCrc[1]);
    Serial.println();

    Serial.print(F("Received CRC  : "));
    printHex(receivedCrcLow);
    Serial.print(' ');
    printHex(receivedCrcHigh);
    Serial.println();

    return calculatedCrc[0] == receivedCrcLow &&
           calculatedCrc[1] == receivedCrcHigh;
}

//-----------------------------------------------------
// Read response in chunks
//-----------------------------------------------------
bool readResponse(uint8_t *response,
                  size_t responseCapacity,
                  size_t &responseLength,
                  uint16_t pollingTimeoutMs)
{
    responseLength = 0;

    uint8_t count = 0;
    bool receivedCount = false;

    const unsigned long startTime = millis();

    /*
       Poll until the ATECC acknowledges the read.
       The device NACKs while it is busy.
    */
    while ((millis() - startTime) < pollingTimeoutMs)
    {
        uint8_t requested =
            Wire.requestFrom((uint8_t)ATECC_ADDR, (uint8_t)1);

        if (requested == 1 && Wire.available())
        {
            count = Wire.read();
            receivedCount = true;
            break;
        }

        while (Wire.available())
        {
            Wire.read();
        }

        delay(1);
    }

    if (!receivedCount)
    {
        Serial.println(F("Timed out waiting for response."));
        return false;
    }

    if (count < 4)
    {
        Serial.print(F("Invalid response count: "));
        Serial.println(count);
        return false;
    }

    if (count > responseCapacity)
    {
        Serial.print(F("Response too large: "));
        Serial.println(count);

        Serial.print(F("Buffer capacity: "));
        Serial.println(responseCapacity);

        return false;
    }

    response[0] = count;

    size_t index = 1;
    size_t remaining = count - 1;

    while (remaining > 0)
    {
        const uint8_t requestedChunk =
            (remaining > WIRE_BUFFER_SIZE)
                ? WIRE_BUFFER_SIZE
                : (uint8_t)remaining;

        const uint8_t receivedChunk =
            Wire.requestFrom(
                (uint8_t)ATECC_ADDR,
                requestedChunk
            );

        if (receivedChunk == 0)
        {
            Serial.println(F("No bytes received during response."));
            return false;
        }

        uint8_t copied = 0;

        while (Wire.available() &&
               copied < receivedChunk &&
               index < responseCapacity)
        {
            response[index++] = Wire.read();
            copied++;
        }

        if (copied != requestedChunk)
        {
            Serial.print(F("Partial response read. Requested "));
            Serial.print(requestedChunk);
            Serial.print(F(", received "));
            Serial.println(copied);

            return false;
        }

        /*
           Subtract bytes actually copied, not bytes merely requested.
        */
        remaining -= copied;
    }

    responseLength = index;

    Serial.print(F("Response length: "));
    Serial.println(responseLength);

    Serial.print(F("Response       : "));
    printBuffer(response, responseLength);

    if (responseLength != count)
    {
        Serial.println(F("Response count mismatch."));
        return false;
    }

    if (!validateResponseCrc(response, responseLength))
    {
        Serial.println(F("Response CRC FAILED."));
        return false;
    }

    Serial.println(F("Response CRC OK."));

    if (responseLength == 4)
    {
        printStatusCode(response[1]);

        return response[1] == 0x00;
    }

    return true;
}

//-----------------------------------------------------

bool sendCommand(uint8_t opcode,
                 uint8_t param1,
                 uint16_t param2,
                 const __FlashStringHelper *name,
                 uint16_t timeoutMs = 100)
{
    uint8_t packet[8];

    buildCommand(opcode, param1, param2, packet);

    Serial.println(F("--------------------------------"));
    Serial.print(F("Sending "));
    Serial.println(name);

    Serial.print(F("Packet: "));
    printBuffer(packet, sizeof(packet));

    Wire.beginTransmission(ATECC_ADDR);
    Wire.write(packet, sizeof(packet));

    const uint8_t txStatus = Wire.endTransmission();

    Serial.print(F("TX status: "));
    Serial.println(txStatus);

    if (txStatus != 0)
    {
        Serial.println(F("Command transmission failed."));
        return false;
    }

    uint8_t response[ATECC_MAX_RESPONSE];
    size_t responseLength = 0;

    return readResponse(
        response,
        sizeof(response),
        responseLength,
        timeoutMs
    );
}

//-----------------------------------------------------

void setup()
{
    Serial.begin(9600);

    while (!Serial)
    {
    }

    Wire.begin();
    Wire.setClock(100000);

    delay(1000);

    Serial.println(F("========================================"));
    Serial.println(F("ATECC608B Communication Demonstration"));
    Serial.println(F("========================================"));
}

void loop()
{
    Serial.println();
    Serial.println(F("========================================"));
    Serial.println(F("New Test Cycle"));
    Serial.println(F("========================================"));

    if (!wakeChip())
    {
        Serial.println(F("Wake failed."));
        delay(3000);
        return;
    }

    Serial.println();

    // INFO / Revision
    sendCommand(
        0x30,
        0x00,
        0x0000,
        F("INFO"),
        100
    );

    Serial.println();

    // RANDOM / standard mode
    sendCommand(
        0x1B,
        0x00,
        0x0000,
        F("RANDOM"),
        100
    );

    Serial.println();
    Serial.println(F("Sleep"));

    sleepChip();

    Serial.println(F("Chip sleeping."));
    Serial.println(F("Waiting 3 seconds..."));

    delay(3000);
}
